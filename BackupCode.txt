        ptpOrderHistory(state, value) {
            state.ptpOrderHistory = value;

            var ptpOrderData = {
                columns: [
                    {
                        label: "Reference",
                        field: "reference",
                        sort: "asc"
                    },
                    {
                        label: "Created",
                        field: "created",
                        sort: "asc"
                    },
                    {
                        label: "Link",
                        field: "link",
                        sort: "asc"
                    },
                    {
                        label: "Is this time based ?",
                        field: "isTimeBased",
                        sort: "asc"
                    },
                    {
                        label: "Credit",
                        field: "credit",
                        sort: "asc"
                    },
                    {
                        label: "Hits Received",
                        field: "hitsReceived",
                        sort: "asc"
                    }
                ],
                rows: []
            };

            for (var i = 0; i < Object.keys(value).length; i++) {
                ptpOrderData.rows.push({
                    reference: value[i].Reference,
                    created: moment(value[i].Creation).format("YYYY-MM-DD - hh:mm:ss A"),
                    link: value[i].Link,
                    isTimeBased: value[i].IsTimeBased.toString(),
                    credit: value[i].Credit,
                    hitsReceived: value[i].HitsReceived
                });
            }

            state.reactivePTPOrderData = ptpOrderData;
        },
